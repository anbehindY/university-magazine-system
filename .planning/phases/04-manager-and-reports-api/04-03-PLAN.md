const submissions = await prisma.submission.findMany({
  where: { isSelected: true },

**ZIP assembly with archiver (CRITICAL — serial streaming, never Promise.all):**

```typescript
const archive = archiver("zip", { zlib: { level: 6 } });

// Error handling: archive errors should not crash the process
archive.on("error", (err) => {
  console.error("Archive error:", err);
});

// IIFE for serial blob fetching — must complete before finalize()
(async () => {
  try {
    for (const submission of submissions) {
      const facultyName = facultyMap.get(submission.facultyId ?? "") ?? "Unknown";
      const studentName = submission.user?.name ?? "Unknown";
      for (const file of submission.files) {
        try {
          const res = await fetch(file.url);
          if (!res.ok || !res.body) {
            // Error handling per Claude's discretion: skip failed fetches, append error marker
            console.error(`Failed to fetch blob: ${file.url} (${res.status})`);
            archive.append(`Failed to download file: ${file.url} (HTTP ${res.status})`, {
              name: `${facultyName}/${studentName}/${file.pathname.split("/").pop() ?? file.id}.error.txt`,
            });
            continue;
          }
          const filename = file.pathname.split("/").pop() ?? file.id;
          archive.append(Readable.fromWeb(res.body as ReadableStream), {
            name: `${facultyName}/${studentName}/${filename}`,
          });
        } catch (err) {
          console.error(`Error fetching blob ${file.url}:`, err);
          archive.append(`Error downloading file: ${String(err)}`, {
            name: `${facultyName}/${studentName}/${file.pathname.split("/").pop() ?? file.id}.error.txt`,
          });
        }
      }
    }

    // Add empty faculty folders for faculties with no selected submissions
    const selectedFacultyIds = new Set(
      submissions.map(s => s.facultyId).filter(Boolean)
    );
    for (const [fId, fName] of facultyMap) {
      if (!selectedFacultyIds.has(fId)) {
        archive.append("", { name: `${fName}/.gitkeep` });
      }
    }

    archive.finalize();
  } catch (err) {
    console.error("Archive assembly error:", err);
    archive.abort();
  }
})();
```

**Stream bridging and response:**

```typescript
// Bridge Node.js stream to Web ReadableStream — Node.js v24 provides Readable.toWeb natively
const webStream = Readable.toWeb(archive) as ReadableStream;

// ZIP filename uses active academic year label if available
const activeYear = await getActiveAcademicYear();
const yearLabel = activeYear?.title ?? "";
const zipFilename = yearLabel
  ? `selected-submissions-${yearLabel}.zip`
  : "selected-submissions.zip";

return new Response(webStream, {
  headers: {
    "Content-Type": "application/zip",
    "Content-Disposition": `attachment; filename="${zipFilename}"`,
  },
});
```

**ANTI-PATTERNS — do NOT do these (per STATE.md and RESEARCH.md):**
- Never `Promise.all()` blob fetches — serial only, prevents memory exhaustion
- Never buffer the full ZIP in memory — stream directly through Response
- Never use `getDownloadUrl()` from `@vercel/blob` — use `fetch(file.url)` directly
- The IIFE must start BEFORE `Readable.toWeb()` is called but after `archive` is created — archiver buffers internally and the web stream reads as data becomes available

**NOTE on getActiveAcademicYear call:** This is called AFTER the streaming response setup is initiated, purely for the filename. Move it before the IIFE if needed. The key constraint is that `Readable.toWeb(archive)` and `new Response(webStream)` must happen synchronously after archive creation — the IIFE runs asynchronously and feeds into the archive.
