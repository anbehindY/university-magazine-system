In `lib/auth-client.ts`, add the `inferAdditionalFields` client plugin so that `useSession()` returns properly typed custom fields instead of requiring `(user as any)` casts:

```typescript
import { adminClient, inferAdditionalFields } from "better-auth/client/plugins";
import { createAuthClient } from "better-auth/react";
import type { auth } from "@/lib/auth";

export const { getSession, signIn, signOut, useSession } = createAuthClient({
  plugins: [
    adminClient(),
    inferAdditionalFields<typeof auth>(),
  ],
});
```

This uses the monorepo/single-project approach where `typeof auth` infers the additionalFields config from the server.
