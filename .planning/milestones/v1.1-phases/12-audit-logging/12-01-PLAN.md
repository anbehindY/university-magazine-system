  <files>app/api/admin/audit-log/route.ts</files>
  <action>
Create `app/api/admin/audit-log/route.ts` — a GET endpoint accessible to ADMINISTRATOR and MARKETING_MANAGER roles.

Follow the exact pagination pattern from `app/api/admin/users/route.ts`:

```typescript
import { requireRole } from "@/lib/auth-helpers";
import prisma from "@/lib/prisma";
import { NextRequest, NextResponse } from "next/server";

export async function GET(req: NextRequest) {
  try {
    const { authorized, error } = await requireRole(["ADMINISTRATOR", "MARKETING_MANAGER"]);
    if (!authorized) {
      return NextResponse.json({ error: error ?? "Unauthorized" }, { status: 403 });
    }

    const { searchParams } = new URL(req.url);
    const page = Math.max(1, parseInt(searchParams.get("page") ?? "1", 10) || 1);
    const pageSize = (() => {
      const raw = parseInt(searchParams.get("pageSize") ?? "10", 10);
      return [10, 25, 50].includes(raw) ? raw : 10;
    })();
    const skip = (page - 1) * pageSize;

    // Date filtering
    const from = searchParams.get("from");
    const to = searchParams.get("to");

    const where: any = {};
    if (from || to) {
      where.createdAt = {};
      if (from) where.createdAt.gte = new Date(from);
      if (to) {
        // Set to end of day for "to" date
        const toDate = new Date(to);
        toDate.setHours(23, 59, 59, 999);
        where.createdAt.lte = toDate;
      }
    }

    const [total, entries] = await Promise.all([
      prisma.auditLog.count({ where }),
      prisma.auditLog.findMany({
        where,
        skip,
        take: pageSize,
        orderBy: { createdAt: "desc" },
        select: {
          id: true,
          action: true,
          entityType: true,
          entityId: true,
          oldValue: true,
          newValue: true,
          metadata: true,
          createdAt: true,
          actor: {
            select: {
              id: true,
              name: true,
            },
          },
        },
      }),
    ]);

    return NextResponse.json({ entries, total, page, pageSize });
  } catch (error) {
    console.error("Error fetching audit log:", error);
    return NextResponse.json(
      { error: "Internal server error" },
      { status: 500 }
    );
  }
}
```

Key points:
- Uses `requireRole()` (which includes mustChangePassword check from Phase 11)
- Parallel `count()` + `findMany()` for efficient pagination
- Date filtering via `from` and `to` query params (ISO date strings)
- `to` date includes the full day (set to 23:59:59.999)
- Orders by `createdAt: "desc"` (newest first)
- Includes `actor` relation for coordinator name (could also use metadata, but relation is more reliable)
- Returns `{ entries, total, page, pageSize }`

function getDateRange(preset: string): { from: string; to: string } | null {
  if (preset === "all") return null;
  const now = new Date();
  const to = now.toISOString().split("T")[0];
  if (preset === "today") return { from: to, to };
  if (preset === "7d") {
    const d = new Date(now);
    d.setDate(d.getDate() - 7);
    return { from: d.toISOString().split("T")[0], to };
  }
  if (preset === "30d") {
    const d = new Date(now);
    d.setDate(d.getDate() - 30);
    return { from: d.toISOString().split("T")[0], to };
  }
  return null;
}
